codice INT primary key,

nota VARCHAR(200) not null,

-- Creazione delle nuove tabelle richieste dal progetto

-- Tabella Volontario
CREATE TABLE Volontario (
    cf VARCHAR(16) PRIMARY KEY,
    nome VARCHAR(50) NOT NULL,
    cognome VARCHAR(50) NOT NULL,
    nascita DATE NOT NULL
);

-- Tabella Associazione
CREATE TABLE Associazione (
    nome VARCHAR(50) PRIMARY KEY
);

-- Tabella Volontariato (Relazione tra Volontario e Associazione)
CREATE TABLE Volontariato (
    volontario VARCHAR(16),
    associazione VARCHAR(50),
    FOREIGN KEY (volontario) REFERENCES Volontario(cf),
    FOREIGN KEY (associazione) REFERENCES Associazione(nome),
    PRIMARY KEY (volontario, associazione)
);

-- Tabella Attestato
CREATE TABLE Attestato (
    codice INT PRIMARY KEY,
    volontario VARCHAR(16) NOT NULL,
    tipo VARCHAR(50) NOT NULL,
    dataAcquisizione DATE NOT NULL,
    FOREIGN KEY (volontario) REFERENCES Volontario(cf)
);

-- Tabella Presenza (Rappresenta le presenze dei volontari nei reparti)
CREATE TABLE Presenza (
    volontario VARCHAR(16),
    nomeReparto VARCHAR(100),
    codiceOspedale INT,
    durata INT NOT NULL,
    FOREIGN KEY (volontario) REFERENCES Volontario(cf),
    FOREIGN KEY (nomeReparto, codiceOspedale) REFERENCES Reparto(nome, codiceospedale),
    PRIMARY KEY (volontario, nomeReparto, codiceOspedale)
);

-- Tabella Attivita
CREATE TABLE Attività (
    codice INT PRIMARY KEY,
    nomeReparto VARCHAR(100) NOT NULL,
    codiceOspedale INT NOT NULL,
    volontario VARCHAR(16) NOT NULL,
    descrizione VARCHAR(200),
    FOREIGN KEY (nomeReparto, codiceOspedale) REFERENCES Reparto(nome, codiceospedale),
    FOREIGN KEY (volontario) REFERENCES Volontario(cf)
);

-- Tabella Seguire (Relazione tra medico, volontario e attività)
CREATE TABLE Seguire (
    volontario VARCHAR(16),
    medico VARCHAR(16),
    attivita INT,
    FOREIGN KEY (volontario) REFERENCES Volontario(cf),
    FOREIGN KEY (medico) REFERENCES Medico(cfimpiegato),
    FOREIGN KEY (attivita) REFERENCES Attività(codice),
    PRIMARY KEY (volontario, medico, attivita)
);

-- Query 1: Determinare il numero di presenze di ogni volontario per reparto
SELECT 
    p.nomeReparto, 
    p.codiceOspedale, 
    COUNT(p.volontario) AS numero_presenze
FROM Presenza p
GROUP BY p.nomeReparto, p.codiceOspedale;

-- Query 2: Determinare i medici che hanno seguito almeno 5 volontari
SELECT 
    s.medico, 
    COUNT(DISTINCT s.volontario) AS numero_volontari
FROM Seguire s
GROUP BY s.medico
HAVING COUNT(DISTINCT s.volontario) >= 5;

SELECT 
    v.cf AS codice_fiscale,
    v.nome AS nome_volontario,
    v.cognome AS cognome_volontario,
    v.nascita AS data_nascita,
    a.nomeReparto AS reparto,
    a.codiceOspedale AS ospedale,
    a.descrizione AS descrizione_attivita
FROM 
    Volontario v
JOIN 
    Attivita a ON v.cf = a.volontario
WHERE 
    a.descrizione = 'Supporto amministrativo';
