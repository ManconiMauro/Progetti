	Scan(&età2)

    var  media, media2, mediaD, mediaE float64

    somma=età1+età2;
    Println("La somma delle 2 età è:",somma)
    media=float64(somma)/2
    Println("La media aritmetica delle 2 età è:",media)
    mediaD = math.Floor(media)
    Println("La media per difetto delle 2 età è:",mediaD)
    mediaE= math.Ceil(media)
    Println("La media per eccesso delle 2 età è:",mediaE)
