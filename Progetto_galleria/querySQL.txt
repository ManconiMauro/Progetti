
create table Autore(
codiceFiscale varchar(16) PRIMARY KEY,
nome varchar(20),
cognome varchar(20)
);


create table Opera(
id integer PRIMARY KEY,
titolo varchar(30),
AnnoComposizione DATE,
prezzo double,
stato bool,
autore varchar(16) NOT NULL,
FOREIGN KEY (autore) REFENCES Autore (codiceFiscale)
);

create table Utente(
id integer PRIMARY KEY,
nome varchar(20),
cognome varchar(20),
nomeUtente varchar(50),
password varchar(50),
numeroCartaDiCredito varchar(20),
codiceCarta varchar(3)
);

create table Ordine(
id integer PRIMARY KEY,
dataAcquisto DATE,
statoOrdine varchar(10),
idOpera integer NOT NULL,
idUtente integer NOT NULL,
FOREIGN KEY (idOpera) REFERENCES Opera (id),
FOREIGN KEY (idUtente) REFERENCES Utente(id)
);
